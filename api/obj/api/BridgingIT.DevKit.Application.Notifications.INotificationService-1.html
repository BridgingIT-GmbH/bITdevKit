Sends or queues notification messages of the specified type.

public interface INotificationService<TMessage> where TMessage : class, INotificationMessage

Type Parameters

TMessage

The notification message type.

Examples

var result = await notificationService.SendAsync(
    new EmailMessage
    {
        Id = Guid.NewGuid(),
        To = ["alice@example.com"],
        Subject = "Todo created",
        Body = "Your todo item is ready."
    },
    new NotificationSendOptions { SendImmediately = true },
    cancellationToken);
