Examples

public class SampleOrchestration : Orchestration<SampleData>
{
    protected override void Define(IOrchestrationBuilder<SampleData> builder)
    {
        builder
            .State("Start", state => state
                .Activity((context, cancellationToken) =>
                {
                    context.Data.Value++;
                    return Task.FromResult(OrchestrationOutcome.Continue());
                })
                .TransitionTo("Done"))
            .State("Done", state => state.Complete());
    }
}
