AddFileMonitoring(IServiceCollection, Action<FileMonitoringBuilder>, ServiceLifetime)

Adds the FileMonitoring system to the service collection with a fluent configuration API. Registers FileMonitoringService as a singleton and allows customization via a builder.

public static FileMonitoringBuilderContext AddFileMonitoring(this IServiceCollection services, Action<FileMonitoringBuilder> configure, ServiceLifetime lifetime = ServiceLifetime.Scoped)

Parameters

services IServiceCollection

The IServiceCollection to add services to.

configure Action<FileMonitoringBuilder>

An action to configure the FileMonitoringBuilder.

lifetime ServiceLifetime

The lifetime used for the monitoring service and default event store.

Returns

FileMonitoringBuilderContext

A FileMonitoringBuilderContext for further configuration chaining.

Examples

services.AddFileMonitoring(monitoring =>
{
    monitoring
        .WithBehavior<LoggingBehavior>() // monitoring behavior for all locations
        .UseLocal("Docs", "C:\\Docs", options =>
        {
            options.FilePattern = "*.txt";
            options.UseOnDemandOnly();
            options.WithProcessorBehavior<LoggingProcessorBehavior>();
            options.UseProcessor<FileMoverProcessor>(config =>
                config.WithConfiguration(p => ((FileMoverProcessor)p).DestinationRoot = "C:\\MovedDocs")) // configure processor instance
                .WithBehavior<RetryProcessorBehavior>();
         })
        .UseLocal("OtherDocs", "C:\\OtherDocs", options =>
        {
            options.FilePattern = "*.pdf";
            options.UseProcessor<FileLoggerProcessor>();
            options.UseProcessor<FileMoverProcessor>(config =>
                config.WithConfiguration(p => ((FileMoverProcessor)p).DestinationRoot = "C:\\OtherMovedDocs")); // configure processor instance
        });
})
.WithEntityFrameworkStore<MyAppDbContext>(); // Extension from EF infra project
