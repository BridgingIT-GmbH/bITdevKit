using BridgingIT.DevKit.Common;
using BridgingIT.DevKit.Presentation.Web;
using BridgingIT.DevKit.Presentation.Web.Dashboard;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddProfiling(options => options.Enabled()).WithRuntimeProfiling().WithOperationProfiling();
builder.Services.AddDashboard(options => options.AllowAnonymous().WithGroupPath("/admin"));
var app = builder.Build();
app.MapEndpoints();
app.MapGet("/fixture/seed", async (IOperationProfiler profiler, OperationProfilingWriterService writer) =>
{
    Guid id;
    using (var operation = profiler.BeginOperation("catalog:refresh", OperationProfilingKind.HttpRequest))
    {
        id = operation.Id;
        operation.SetHttpMetadata(new() { Method = "GET", StatusCode = 200 });
        operation.SetDimension("region", "west");
        for (var index = 0; index < 2; index++)
        {
            using var segment = profiler.BeginSegment("Load");
            await Task.Delay(120);
            using (var child = profiler.BeginSegment("Compute")) { await Task.Delay(60); child.Complete(); }
            segment.Complete();
        }
        await Task.WhenAll(Enumerable.Range(0, 2).Select(async _ => { using var segment = profiler.BeginSegment("ParallelWork"); await Task.Delay(100); segment.Complete(); }));
        operation.Complete();
    }
    await writer.TickAsync();
    return Results.Ok(id);
});
await app.StartAsync();
var store = app.Services.GetRequiredService<IRuntimeProfilingStore>();
var node = app.Services.GetRequiredService<IProfilingNodeIdentityProvider>().GetNode();
var correlation = new RuntimeProfilingNodeCorrelation(Guid.NewGuid().ToString("N"), node.ProcessStartedUtc);
_ = (await store.GetOrCreateNodeAsync(correlation, node with { Correlation = correlation }));
var started = DateTimeOffset.UtcNow;
var session = (await store.GetOrCreateActiveSessionAsync(new(ProfilingIdentityFactory.CreateRuntimeSession(), "Browser evidence", started, TimeSpan.FromMilliseconds(500), TimeSpan.FromMinutes(20), []))).Value.Session;
_ = (await store.UpsertParticipationAsync(new() { SessionId = session.Identity.Id, SessionKey = session.Identity.Key, NodeId = node.Identity.Id, NodeKey = node.Identity.Key, JoinedUtc = started, Role = RuntimeProfilingNodeRole.ExpectedParticipant, State = RuntimeProfilingParticipationState.Collecting }));
var sampling = Task.Run(async () =>
{
    for (var index = 1L; index <= 1200; index++)
    {
        _ = (await store.AddSnapshotAsync(new() { Identity = ProfilingIdentityFactory.CreateRuntimeSnapshot(), SessionId = session.Identity.Id, SessionKey = session.Identity.Key, NodeId = node.Identity.Id, NodeKey = node.Identity.Key, Sequence = index, TimestampUtc = started.AddMilliseconds(index * 500), CaptureStartedElapsed = TimeSpan.FromMilliseconds(index * 500), CpuUsagePercent = index % 70, WorkingSetBytes = 60000000 + index * 10000, ManagedMemoryBytes = 20000000, AllocationRateBytesPerSecond = 20000, Gen0CollectionCount = index / 8 }));
        await Task.Delay(500);
    }
});
await app.WaitForShutdownAsync();
